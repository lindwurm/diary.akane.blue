{{- /* Hexo の toc(page.content, {list_number: false}) 相当。 render-heading フックが出力した見出しを本文から拾い、Hexo と同じ入れ子の
    を組み立てる。 */ -}} {{- $re := `(.*?)` -}} {{- $headings := findRESubmatch $re (string .) -}} {{- $out := "" -}} {{- if $headings -}} {{- $out = `
      ` -}} {{- $stack := slice -}} {{- range $headings -}} {{- $level := int (index . 1) -}} {{- $id := index . 2 -}} {{- $text := index . 3 | plainify -}} {{- $popped := false -}} {{- range $stack -}} {{- if and (gt (len $stack) 0) (ge (index $stack (sub (len $stack) 1)) $level) -}} {{- $out = printf "%s" $out -}} {{- $stack = first (sub (len $stack) 1) $stack -}} {{- $popped = true -}} {{- if and (gt (len $stack) 0) (ge (index $stack (sub (len $stack) 1)) $level) -}} {{- $out = printf "%s
    " $out -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (gt (len $stack) 0) (not $popped) -}} {{- $out = printf `%s
      ` $out -}} {{- end -}} {{- $out = printf `%s
    1. %s` $out $level (urlquery $id) $text -}} {{- $stack = $stack | append $level -}} {{- end -}} {{- range $i, $_ := $stack -}} {{- $out = printf "%s
      2. " $out -}} {{- if lt $i (sub (len $stack) 1) }}{{ $out = printf "%s
    " $out }}{{ end -}} {{- end -}} {{- $out = printf "%s
" $out -}} {{- end -}} {{- return ($out | safeHTML) -}}
